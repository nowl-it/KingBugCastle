# Server runtime. Install into a venv, not the system Python:
#
#   python -m venv .venv
#   python -m pip install -r requirements.txt  # after activating the venv
#
# server/run.py finds .venv/bin on macOS/Linux and .venv/Scripts on Windows.
# Distro Pythons (Arch, Debian) refuse a system-wide `pip install` anyway, and upgrading FastAPI
# in /usr would move it for every other project on the machine.
#
# Pinned to a floor, not an exact version: these are the versions this server is
# tested against, and anything newer in the same line has been fine.

gunicorn>=21.2.0; sys_platform != "win32"  # Production process manager; Unix only
watchfiles>=1.2.0             # run.py --reload; without it uvicorn silently falls back to
                              # StatReload, watches only .py, and ignores --reload-include
windows-curses>=2.4.1; sys_platform == "win32"  # TUI support on Windows
