# Server runtime. Install into a venv, not the system Python:
#
#   python3 -m venv .venv && .venv/bin/pip install -r requirements.txt
#
# server/run.sh picks .venv/bin up automatically when it exists. Distro Pythons
# (Arch, Debian) refuse a system-wide `pip install` anyway, and upgrading FastAPI
# in /usr would move it for every other project on the machine.
#

# tested against, and anything newer in the same line has been fine.
